\begin{theorem}[Source restriction and minimal lift]
Let \(S(q)P\) be a rich split representation and let \(B\) be a consumed source restriction.  Put
\[
A(q)=B S(q)P,
\qquad

\]
and the blind kernel is
\[
\bigcap_s\Ker(BD_sP).
\]
If scalar supplemental observations \(G\) are added, the augmented map is injective exactly when
\[
\Ker A\cap\Ker G=\{0\}.
\]

The rank formula is the rank of the direct sum of all coefficient-level source restrictions. This gives the injectivity criterion and the lower bound on the number of independent scalar repairs.
\end{proof}

\section{Positive-eta amplitude analyticity}

The positive-eta analytic theorem is one of the load-bearing results.  It removes path-order ambiguity before the sign problem begins.

\begin{definition}[Polar frame transport]
For a full-column-rank amplitude \(\Theta=QH\), where \(Q^*Q=I_5\) and \(H>0\), define the canonical transport between two polar frames \(Q_a,Q_b\) by
\[
U_{a\to b}=\polar(Q_b^*Q_a).
\]
\end{definition}

\begin{theorem}[Fixed-five positive-eta path equality]
Let
\[
A_{\rm chart}=AU,
\qquad
\Theta_\eta=Q_{\rm chart}G_\eta^{1/2},
\qquad
G_\eta=1\oplus(A_+ + \eta I_4),\quad \eta>0.
\]
Here \(A\) is the eta-independent raw amplitude, \(U\) is the fixed unitary accepted chart, \(Q_{\rm chart}\) is the polar orientation of \(A_{\rm chart}\), and \(G_\eta^{1/2}\) is strictly positive.  Then
\[
\polar(\Theta_\eta)=Q_{\rm chart}
\]
for every positive eta.  Hence, for any positive eta values \(\eta_1,\eta_2\),
\[
\text{raw eta transport}=I_5,
\qquad
\text{normalized eta transport}=I_5,
\qquad
\text{raw-to-normalized transport}=U^*.
\]
All positive-eta normalization/continuation rectangles are identity loops on the fixed five-label carrier.
\end{theorem}

\begin{proof}
Right multiplication by a unitary is polar-covariant, so \(\polar(AU)=\polar(A)U\).  Right multiplication by a strictly positive factor changes the metric scale but not the polar orientation.  Since \(G_\eta^{1/2}>0\), the polar orientation of \(Q_{\rm chart}G_\eta^{1/2}\) is \(Q_{\rm chart}\).  Therefore the polar frame at every normalized positive-eta node is the same frame.  The raw eta transport and normalized eta transport are identity maps on the label fibre, while the raw-to-normalized transport is the fixed chart unitary \(U^*\).  The two possible paths around any positive-eta normalization rectangle have the same product.
\end{proof}

\section{Zero as cut}

The previous theorem says there is no positive-eta normalization holonomy.  The endpoint issue is therefore the support event at eta zero.  This is the role of the zero-as-cut theorem.

\begin{definition}[Cut and joint cycles]
The pre-cut carrier has one geometric cycle \(\gamma_\pi\).  The post-cut carrier has an independent cut-memory cycle:
\[
H_1(S^1;\mathbb Z)=\mathbb Z\gamma_\pi,
\qquad
H_1(T^2;\mathbb Z)=\mathbb Z\gamma_\pi\oplus\mathbb Z\mu_{\rm cut}.
\]
On the ordered basis \((\gamma_\pi,\mu_{\rm cut})\), set
\[
P_{\rm joint}=\diag(1,0),
\qquad
P_{\rm cut}=\diag(0,1).
\]
\end{definition}

\begin{theorem}[Zero-as-cut projector theorem]
The cut and joint projectors satisfy
\[
P_{\rm cut}P_{\rm joint}=P_{\rm joint}P_{\rm cut}=0,
\qquad
P_{\rm cut}+P_{\rm joint}=I.
\]
Hence no nonzero vector can be both a pure cut state and a complete joint state.
\end{theorem}

\begin{proof}
The identities follow immediately from the displayed diagonal matrices.  If \(u=P_{\rm cut}u=P_{\rm joint}u\), then
\[
u=P_{\rm cut}P_{\rm joint}u=0.
\]
Thus simultaneous pure cut recognition and complete joint recognition force vanishing.
\end{proof}

\begin{remark}[Visible return is not full return]
For the join projection \(J=(1,0)\) and section \(S=(1,0)^T\), one has \(JS=I_1\), but
\[
SJ=P_{\rm joint}\ne I_2.
\]
Thus the visible circle coordinate can return while the full post-cut torus state does not.  The exact memory defect is
\[
I_2-SJ=P_{\rm cut}.
\]
This is the reason the endpoint is handled by a zero-cut source-Gram theorem rather than by an ordinary positive-eta continuation.
\end{remark}

\begin{remark}[Finite critical witness]
The finite critical response packet recorded a surviving cut-memory coordinate with joint and cut-memory component norms both \(4.137400474477719\cdot10^{-5}\), relative cut memory \(0.7071067811865472\), and complete state return false.  This is a finite witness for the warning above: corrected target visibility does not automatically give full state return.
\end{remark}

\section{Seam integer and the common five-matrix}

The sign-sensitive obstruction is represented by one rank-at-most-five matrix.  This section turns the seam language into a standard threshold-count statement.

\begin{definition}[Seam charge]
Let \(K_\eta\) be the compact seam response at positive shift eta.  Define
\[
\kSigma(\eta)=\rank\mathbf 1_{(1,\infty)}(K_\eta).
\]
On the five-label source packet this is represented by
\[
\BSEta=V_\eta^*S_{\eta,-}^{-1}V_\eta,
\qquad
\rank V_\eta\le5,
\]
and
\[
\kSigma(\eta)=N_+(\BSEta-I).
\]
\end{definition}

\begin{theorem}[Common five-matrix covariance]
For a positive source operator \(S\) and a five-column source map \(V\), set
\[
B=V^*S^{-1}V\in M_5(\C).
\]
If \(S^U=USU^*\) and \(V^U=UV\), then
\[
(V^U)^*(S^U)^{-1}V^U=B.
\]
A unitary change of the five labels conjugates \(B\).  Therefore the spectrum, threshold projection, seam charge and sign decision are representation invariant.
\end{theorem}

\begin{proof}
Since \((USU^*)^{-1}=US^{-1}U^*\),
\[
(V^U)^*(S^U)^{-1}V^U
=V^*U^*US^{-1}U^*UV
=V^*S^{-1}V.
\]
If the label basis is changed by a unitary \(U_5\), then \(V\) is replaced by \(VU_5\), and the matrix becomes \(U_5^*BU_5\).  Hence threshold ranks and spectral projections are invariant.
\end{proof}

\begin{corollary}[Matrix, seam and threshold holonomy]
The native matrix, prime-torus matrix and threshold-holonomy presentation are faithful avatars of the same obstruction:
\[
\BSEta\le I
\iff
\kSigma(\eta)=0
\iff
H_{\rm threshold}(\eta)=I,
\]
where
\[
H_{\rm threshold}(\eta)=I-2\mathbf 1_{(1,\infty)}(\BSEta).
\]
Thus the prime torus is a chart of the same matrix, not a second independent proof route.
\end{corollary}

\section{The accepted block and \texorpdfstring{$M_3$}{M3}}

The finite accepted odd block is

\lambda_{\min}(G_3)=2.0838177120437723\cdot10^{-13}>0

Therefore \(M_3>0\).  Since \(M_3\) is the Schur complement of \(G_3\) in the bordered Gram, the completed-Weil form is positive definite on \(E_3\).

\section{Zero-cut T03 endpoint completion}

The T03 theorem is the main endpoint route used in the paper.  It replaces orientation-dependent ambient equality by an invariant source-Gram identity.

\begin{definition}[Endpoint source-Gram data]
Let \(\RSigma\) be the total five-label response readout, \(\Psrc\) the source projection, \(\Sminus\) the complete unshifted odd source, and \(\WSigma\) the five-label response map.  The endpoint source-Gram target is
\[
\RSigma^*\Psrc\RSigma
=
\bigl((\Sminus)^{\dagger/2}\WSigma\bigr)^*
\bigl((\Sminus)^{\dagger/2}\WSigma\bigr).
\]
\end{definition}

\begin{theorem}[Six-block convergence criterion]
Suppose finite transported source maps give finite Gram defects \(\Delta_n\) satisfying
\[
\|\Delta_n\|\le
\eps_{\rm vis}(n)+\eps_{\rm blind}(n)+\eps_{\rm cross}(n)
+\eps_{\rm tail}(n)+\eps_{\rm quad}(n)+\eps_{\rm chart}(n),
\]
where the six terms respectively bound visible compression error, five-dimensional blind response error, both cross blocks, omitted prime/Gamma tails, quadrature/window/solve/rounding errors, and native/Feshbach/prime-torus chart transport.  If
\[
\eps_{\rm vis}(n)+\eps_{\rm blind}(n)+\eps_{\rm cross}(n)
+\eps_{\rm tail}(n)+\eps_{\rm quad}(n)+\eps_{\rm chart}(n)\longrightarrow0,
\]
then the invariant source-Gram identity holds.
\end{theorem}

\begin{proof}
Each finite transported source map produces a finite Gram comparison between the total five-label response readout and the source-projected response.  The six listed errors form an outward norm bound on the full Gram defect.  Since the total bound tends to zero, \(\Delta_n\to0\) in operator norm on the finite label space.  Passing to the limit gives equality of the two limiting Gram matrices.  Equal Gram matrices determine a canonical partial isometry between final ranges, so literal equality of ambient source maps is unnecessary.
\end{proof}

\begin{theorem}[Zero-cut projection-defect positivity]

\[
K_0=\RSigma^*(I-\Psrc)\RSigma\ge0.

\end{theorem}

\begin{proof}

The invariant source-Gram identity identifies the source-visible part of the endpoint response with the complete unshifted source readout.  The remaining projection defect is exactly the zero-cut endpoint obstruction.  The Schur endpoint equivalence therefore yields the completed-Weil endpoint sign.
\end{proof}
